## Usage

```tsx
import { MatchTimeline } from "@/components/match-timeline"

<MatchTimeline
  entries={[
    { id: "1", minute: 12, type: "goal", team: "home", description: "Goal — Silva" },
    { id: "2", minute: 34, type: "yellow-card", team: "away", description: "Yellow card — Reyes" },
  ]}
/>
```
