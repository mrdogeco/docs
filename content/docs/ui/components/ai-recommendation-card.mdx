## Usage

```tsx
import { AIRecommendationCard } from "@/components/ai-recommendation-card"

<AIRecommendationCard
  competition="Brasileirão Série A"
  home="Palmeiras"
  away="Corinthians"
  pick="1"
  odds="1.95"
  confidence="high"
  edgePercentage={0.084}
  kellyFraction={0.032}
  rationale={[
    "Palmeiras unbeaten in 8 home matches this season.",
    "Corinthians missing two starting defenders to injury.",
  ]}
  riskFactors={["Palmeiras rotating squad ahead of a continental fixture."]}
/>
```

Once a pick settles, pass `result` — the card swaps the confidence badge
for a [Settlement Badge](/docs/ui/components/settlement-badge) automatically.

## Use with the Mr. Doge SDK

`ai.recommendations.list()`/`.get()` returns `outcome` as a raw code (same
convention as `Line.code` elsewhere in this SDK) rather than a
human-readable label — fully resolving one needs a joined `Market`/`Line`
lookup, out of scope for a single mapping function. This adapter falls back
to the raw code, and requires `match` to be present on the input (request
`select: { match: true }` when fetching) since the card needs the
competition/team names:

```tsx
import type { Recommendation, Match } from "@mrdoge/protocol"
import type {
  AIRecommendationCardProps,
  RecommendationConfidence,
} from "@/components/ai-recommendation-card"

type RecommendationWithMatch = Recommendation & { match: Match }

function toConfidence(confidence: Recommendation["confidence"]): RecommendationConfidence {
  switch (confidence) {
    case "High":
      return "high"
    case "Medium":
      return "medium"
    case "Low":
      return "low"
  }
}

function toPickLabel(rec: Recommendation): string {
  return rec.point != null ? `${rec.outcome} ${rec.point}` : rec.outcome
}

function recommendationToProps(rec: RecommendationWithMatch): AIRecommendationCardProps {
  return {
    competition: rec.match.competition.name,
    home: rec.match.homeTeam.name,
    away: rec.match.awayTeam.name,
    pick: toPickLabel(rec),
    odds: rec.odds.toFixed(2),
    confidence: toConfidence(rec.confidence),
    edgePercentage: rec.edgePercentage,
    kellyFraction: rec.kellyFraction,
    rationale: rec.rationale,
    riskFactors: rec.riskFactors,
    result: rec.settled && rec.result ? rec.result : undefined,
  }
}
```
