## Usage

```tsx
import { TeamFormIndicator } from "@/components/team-form-indicator"

<TeamFormIndicator results={["L", "D", "W", "W", "W"]} />
```

Pass `matches` alongside `results` (same length and order) to make each
pill a tooltip trigger showing the opponent and score — hover a pill in
the preview above to see it.

## Use with the Mr. Doge SDK

`teams.form()` returns both fields most-recent-first; the component wants
oldest-to-newest (most recent last), so both get reversed together to keep
`results[i]`/`matches[i]` pointing at the same game:

```tsx
import type { TeamForm, TeamFormResult } from "@mrdoge/protocol"
import type {
  TeamFormIndicatorProps,
  TeamFormMatchEntry,
  FormResult,
} from "@/components/team-form-indicator"

function toFormResult(result: TeamFormResult): FormResult {
  switch (result) {
    case "win":
      return "W"
    case "draw":
      return "D"
    case "loss":
      return "L"
    case "unknown":
      return "U"
  }
}

function teamFormToProps(form: TeamForm): TeamFormIndicatorProps {
  const matches: TeamFormMatchEntry[] = form.matches
    .map((match) => ({
      opponent: match.opponent.name,
      isHome: match.isHome,
      scoreFor: match.score.for,
      scoreAgainst: match.score.against,
      result: toFormResult(match.result),
    }))
    .reverse()

  return {
    results: form.summary.form.map((code) =>
      code === "W" || code === "D" || code === "L" ? code : "U"
    ).reverse(),
    matches,
  }
}
```
