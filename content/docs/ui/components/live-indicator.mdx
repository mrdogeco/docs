## Use with the Mr. Doge SDK

`status` mostly mirrors the real SDK's `Clock.state` vocabulary directly,
but `match.status` (upcoming/live/completed) is the authoritative field,
always kept fresh by the server — `clock.state` comes from the live-stats
feed, which can stop updating once a match ends (frozen on "live" instead
of ever reaching "finished"). So `match.status` decides the coarse bucket
first; `clock.state` only refines *within* "live":

```tsx
import type { MatchDetail } from "@mrdoge/protocol"
import type { LiveIndicatorStatus } from "@/components/live-indicator"

function toLiveIndicatorStatus(match: MatchDetail): LiveIndicatorStatus {
  if (match.status === "upcoming") return "scheduled"
  if (match.status === "completed") return "finished"
  return match.stats?.clock?.state ?? "live"
}
```
