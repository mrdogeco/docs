## Usage

```tsx
import { MatchTimeline } from "@/components/match-timeline"

<MatchTimeline
  entries={[
    { id: "1", time: "12'", type: "goal", side: "home", description: "Goal — Silva" },
    { id: "2", time: "34'", type: "yellow-card", side: "away", description: "Yellow card — Reyes" },
  ]}
/>
```

## Use with the Mr. Doge SDK

`type` is an open string on both sides — sport-specific event types (the
SDK documents per-sport unions like `SoccerTimelineEventType`,
`TennisTimelineEventType`) pass straight through, and unrecognized types
just fall back to a generic marker instead of being rejected:

```tsx
import type { TimelineEvent } from "@mrdoge/protocol"
import type { MatchTimelineProps } from "@/components/match-timeline"

const MINUTE_CLOCK_PHASES = new Set(["1H", "2H", "ET1", "ET2"])

function toTimeLabel(event: TimelineEvent): string | undefined {
  if (event.timeOffsetSeconds === 0) return undefined
  if (MINUTE_CLOCK_PHASES.has(event.phase)) {
    return `${Math.floor(event.timeOffsetSeconds / 60)}'`
  }
  return event.phase
}

function timelineToMatchTimelineProps(events: TimelineEvent[]): MatchTimelineProps {
  return {
    entries: events.map((event, index) => ({
      id: String(index),
      time: toTimeLabel(event),
      type: event.type,
      side: event.side,
      description: event.captions.join(" — "),
    })),
  }
}
```
