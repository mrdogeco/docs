```tsx
  sports={[

  ]}
  selected="soccer"
  onSelect={(sportId) => console.log(sportId)}
/>
```

## Use with the Mr. Doge SDK

The SDK's `SportName` enum is `snake_case` on the wire
(`"american_football"`, `"ice_hockey"`) — this component's own `Sport` id
is `kebab-case` instead, matching this registry's convention elsewhere
(e.g. Match Timeline's `"yellow-card"`). The conversion table is keyed by
the real enum, so a new sport added to the SDK shows up as a real type
error here instead of silently missing an icon:

```tsx
import type { SportName } from "@mrdoge/protocol"
import type { SportOption } from "@/components/sport-picker"

const SPORT_LABELS: Record<SportName, SportOption> = {
  soccer: { id: "soccer", label: "Soccer" },
  basketball: { id: "basketball", label: "Basketball" },
  american_football: { id: "american-football", label: "American Football" },
  baseball: { id: "baseball", label: "Baseball" },
  ice_hockey: { id: "ice-hockey", label: "Ice Hockey" },
  volleyball: { id: "volleyball", label: "Volleyball" },
  handball: { id: "handball", label: "Handball" },
  tennis: { id: "tennis", label: "Tennis" },
}

function sportNamesToOptions(sports: SportName[]): SportOption[] {
  return sports.map((sport) => SPORT_LABELS[sport])
}
```
