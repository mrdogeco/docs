```tsx

<OddsBoard
  markets={[

      id: "market-1",
      label: "Match Winner",
      options: [
        { id: "home", label: "1", price: "1.85" },
        { id: "draw", label: "X", price: "3.40" },
        { id: "away", label: "2", price: "4.20" },
      ],
    },
    {
      id: "market-2",

      options: [
        { id: "over", label: "Over 2.5", price: "1.95" },
        { id: "under", label: "Under 2.5", price: "1.85" },
      ],
    },
  ]}
/>
```

## Use with the Mr. Doge SDK

`odds.list({ matchId })` returns every live market for a match at once —
this is the actual fix for [Odds Selector](/docs/ui/components/odds-selector)
only handling one market; Odds Board composes several. `Market.betType` is
a raw sysname on the wire (e.g. `"SOCCER_UNDER_OVER"`), not a display
label — a real name lookup is out of scope for a single mapping function,
so this falls back to a light cleanup instead of the raw sysname:

```tsx
import type { Market } from "@mrdoge/protocol"
import type { OddsBoardProps } from "@/components/odds-board"
// toOddsOptions is the same BetItem -> OddsOption mapper shown on the
// Event Card page's SDK example — reuse it instead of duplicating it.
import { toOddsOptions } from "@/lib/sdk-adapters"

function toMarketLabel(betType: string): string {
  return betType
    .toLowerCase()
    .split("_")
    .map((word) => word[0].toUpperCase() + word.slice(1))
    .join(" ")
}

function marketsToOddsBoardProps(markets: Market[]): OddsBoardProps {
  return {
    markets: markets.map((market) => ({
      id: market.id,
      label: toMarketLabel(market.betType),
      options: toOddsOptions(market),
    })),
  }
}
```
